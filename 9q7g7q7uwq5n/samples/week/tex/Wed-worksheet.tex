\begin{center}{\Large \textbf{Answer Key --- fully worked}}\end{center}
\vspace{0.25cm}
\textbf{Procedure:} Denominator $\ne 0$ --- exclude those $x$; Even root's inside must be $\ge 0$ --- solve it; Otherwise the domain is all reals; Range: think about the lowest or highest output.\\[8pt]
\begin{enumerate}[label=\textbf{\arabic*.},itemsep=7pt]
  \item $x\ne 4$
  \item $x\ge -6$
  \item $x\ne \pm 3$ (since $x^{2}-9=0$ at $\pm 3$)
  \item $x\ne -7$
  \item $x\ge 1$
  \item $x\ne \pm 4$ (since $x^{2}-16=0$ at $\pm 4$)
  \item $x\ge 4$ (need $2x-8\ge 0$)
  \item $x\ne 3,\ -2$ (denominator $(x-3)(x+2)=0$)
  \item Domain all reals; range $y\ge -5$
  \item Domain all reals; range $y\ge 2$
  \item $x\ge 1$ and $x\ne 4$ (root needs $x\ge 1$; denominator excludes 4)
\end{enumerate}
