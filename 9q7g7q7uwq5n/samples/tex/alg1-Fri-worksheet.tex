% Factor completely
\textbf{Example (Factor completely):}

$2x^{2}-8$ — GCF is $2$: $2(x^{2}-4)$. Then $x^2-4$ is a difference of squares: $2(x+2)(x-2)$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Factor completely. Always check for a GCF first.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle 3x^{2}-27$
  \item $\displaystyle x^{2}+9x+20$
  \item $\displaystyle 5x^{2}+10x$
  \item $\displaystyle x^{2}-64$
  \item $\displaystyle x^{2}-3x-10$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $4x^{2}-36$ — factor completely (GCF then a pattern).
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Factor out the GCF.
  \item Count the terms: two squares → difference of squares; three terms → product-sum.
  \item Factor the leftover and check by distributing.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item $3(x+3)(x-3)$
  \item $(x+4)(x+5)$
  \item $5x(x+2)$
  \item $(x+8)(x-8)$
  \item $(x-5)(x+2)$
  \item $4(x+3)(x-3)$
\end{enumerate}
