\textbf{Vocabulary}\\[5pt]
\textbf{Vertex Form:} \bl[3.5cm]{$y=a(x-h)^2+k$}, showing the vertex $(h,k)$ directly.\\[10pt]
\textbf{Axis of Symmetry:} The vertical line \bl[2cm]{$x=h$} through the vertex.\\[10pt]
\textbf{Max/Min Value:} The y-coordinate of the vertex, \bl[1cm]{$k$}, occurring at $x=h$.\\[12pt]
\textbf{We do together.}\quad For $y=-2(x+3)^2+6$, find the vertex, y-intercept, transformations, and max/min value.\\[6pt]
\begin{enumerate}[label=\textbf{Step \arabic*.},leftmargin=3.1em,itemsep=9pt]
  \item Vertex: $a=-2$, $h=-3$, $k=6$, so vertex $=$ \bl[2.5cm]{$(-3,6)$}.
  \item y-intercept: $y=-2(0+3)^2+6=-2(9)+6=$ \bl[2cm]{$-12$}, point $(0,-12)$.
  \item Transformations from $y=x^2$: shift left 3, reflect over x-axis, \bl[2cm]{stretch} narrower by 2, shift up 6.
  \item Since $a=-2<0$, this is a \bl[2.5cm]{maximum} value of \bl[1cm]{$6$}, occurring at $x=-3$.
\end{enumerate}\vspace{8pt}
