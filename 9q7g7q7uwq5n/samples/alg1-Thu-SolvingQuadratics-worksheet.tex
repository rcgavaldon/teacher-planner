\begin{center}{\Large \textbf{Solving Quadratics by Factoring}}\end{center}\vspace{0.25cm}
% Solve by factoring
\textbf{Example (Solve by factoring):}

$x^{2}+5x+6=0$. Factor: $(x+2)(x+3)=0$. Set each factor to $0$: $x=-2$ or $x=-3$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Factor, then use the Zero Product Property to solve.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=3cm]
  \item $\displaystyle x^{2}+7x+10=0$
  \item $\displaystyle x^{2}-x-12=0$
  \item $\displaystyle x^{2}-9=0$
  \item $\displaystyle x^{2}-2x-15=0$
  \item $\displaystyle x^{2}+6x=0$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $x^{2}-5x=14$ — first make one side $0$, then solve.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Move all terms to one side so it equals $0$.
  \item Factor the quadratic.
  \item Set each factor $=0$ and solve for $x$.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item $x=-2,\ -5$
  \item $x=4,\ -3$
  \item $x=3,\ -3$
  \item $x=5,\ -3$
  \item $x=0,\ -6$
  \item $x=7,\ -2$
\end{enumerate}
