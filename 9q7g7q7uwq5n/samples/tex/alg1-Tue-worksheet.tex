\begin{center}{\Large \textbf{Factoring Trinomials (x\textasciicircum{}2 + bx + c)}}\end{center}\vspace{0.25cm}
% Factor the trinomial
\textbf{Example (Factor the trinomial):}

$x^{2}+5x+6$ — find two numbers that multiply to $6$ and add to $5$: that is $2$ and $3$. So $x^{2}+5x+6=(x+2)(x+3)$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Factor by finding two numbers that multiply to c and add to b.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle x^{2}+7x+10$
  \item $\displaystyle x^{2}+8x+15$
  \item $\displaystyle x^{2}-5x+6$
  \item $\displaystyle x^{2}+2x-8$
  \item $\displaystyle x^{2}-x-12$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $x^{2}-2x-15$ — factor, then name the values that make it $0$.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item List factor pairs of $c$.
  \item Choose the pair that adds to $b$.
  \item Write $(x+\text{first})(x+\text{second})$ and FOIL to check.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item $(x+2)(x+5)$
  \item $(x+3)(x+5)$
  \item $(x-2)(x-3)$
  \item $(x+4)(x-2)$
  \item $(x-4)(x+3)$
  \item $(x-5)(x+3)$; zeros $x=5,-3$
\end{enumerate}
