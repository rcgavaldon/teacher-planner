\textbf{Vocabulary}\\[5pt]
\textbf{Max / min:} the \bl[2.6cm]{turning point} of a parabola (the vertex); opens up gives a \bl[2cm]{minimum}, opens down gives a \bl[2cm]{maximum}.\\[10pt]
\textbf{Increasing / decreasing:} read left to right --- the graph goes \bl[1.2cm]{up} (increasing) or \bl[1.4cm]{down} (decreasing).\\[12pt]
\textbf{We do together.}\quad For $y=(x-2)^{2}-3$, give the vertex, state max or min, and the intervals.\\[6pt]
\begin{enumerate}[label=\textbf{Step \arabic*.},leftmargin=3.1em,itemsep=9pt]
  \item Vertex form $a(x-h)^{2}+k$ gives vertex \bl[1.8cm]{$(2,-3)$}.
  \item Leading coefficient is $+1$ (opens up), so the vertex is a \bl[1.8cm]{minimum}.
  \item It decreases for \bl[1.5cm]{$x<2$} and increases for \bl[1.5cm]{$x>2$}.
\end{enumerate}\vspace{8pt}
\textbf{See it:}
\begin{center}\begin{tikzpicture}
\begin{axis}[axis lines=middle,xlabel=$x$,ylabel=$y$,xmin=-2,xmax=6,ymin=-4,ymax=7,width=10cm,height=5.2cm,grid=both,samples=80]
\addplot[very thick,blue,domain=-1:5]{(x-2)^2-3};
\addplot[only marks,mark=*,red] coordinates {(2,-3)};
\node[above right,red,font=\small] at (axis cs:2,-3) {min $(2,-3)$};
\end{axis}\end{tikzpicture}\end{center}
{\small\centering opens up $\Rightarrow$ minimum at $(2,-3)$.\par}
