\begin{center}{\Large \textbf{Function Families}}\end{center}\vspace{0.25cm}
% Name the parent function
\textbf{Example (Name the parent function):}

$f(x)=2^{x}$ is the \textbf{exponential} parent. It passes through $(0,1)$, never touches the x-axis, and its domain is all real numbers.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Name the parent family and give the domain.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle f(x)=x^{2}$
  \item $\displaystyle f(x)=|x|$
  \item $\displaystyle f(x)=\dfrac{1}{x}$
  \item $\displaystyle f(x)=\sqrt{x}$
  \item $\displaystyle f(x)=x^{3}$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $f(x)=\log_{2}x$ — name the family and give the domain.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Read the rule and match it to a known family.
  \item Recall the family’s basic shape and key point.
  \item State the domain (watch for square roots and denominators).
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item Quadratic; domain all real numbers
  \item Absolute value; domain all real numbers
  \item Rational (reciprocal); domain $x\neq 0$
  \item Square root; domain $x\ge 0$
  \item Cubic; domain all real numbers
  \item Logarithmic; domain $x>0$
\end{enumerate}
