{\small Algebra II \textbar\ Guided notes --- we fill these in together}\vspace{2pt}\hrule\vspace{7pt}
\textbf{Vocabulary}\\[5pt]
\textbf{Parabola:} The \bl[2cm]{U-shaped} graph produced by a quadratic function.\\[10pt]
\textbf{Vertex:} The turning point of a parabola, written as the ordered pair \bl[2.5cm]{$(h,k)$}.\\[10pt]
\textbf{Vertex Form:} A quadratic equation written as \bl[3.5cm]{$y=a(x-h)^2+k$}, where $(h,k)$ is the vertex.\\[12pt]
\textbf{We do together.}\quad Identify $a$, the vertex, and the direction of opening for $y=2(x-3)^2+1$.\\[6pt]
\begin{enumerate}[label=\textbf{Step \arabic*.},leftmargin=3.1em,itemsep=9pt]
  \item Match the equation to $y=a(x-h)^2+k$: here $a=$ \bl[1cm]{$2$}.
  \item Since the equation has $(x-3)$, we know $h=$ \bl[1cm]{$3$}.
  \item The constant added at the end gives $k=$ \bl[1cm]{$1$}.
  \item So the vertex $(h,k)=$ \bl[2cm]{$(3,1)$}.
  \item Since $a=2$ is \bl[2cm]{positive}, the parabola opens \bl[1.5cm]{up}.
\end{enumerate}\vspace{8pt}

\vspace{6pt}
\textbf{You Try.}\ Solve each. Show every step, then name the answer.\\[6pt]
\begin{enumerate}[label=\textbf{\alph*)},leftmargin=2.4em,itemsep=1.4cm]
  \item Identify $a$, the vertex, and direction of opening for $y=-(x+4)^2+2$. \hfill \ifkey\textcolor{blue!65!black}{$a=-1$, vertex $(-4,2)$, opens down.}\fi
  \item Identify $a$, the vertex, and direction of opening for $y=\frac{1}{2}(x-5)^2-3$. \hfill \ifkey\textcolor{blue!65!black}{$a=\frac{1}{2}$, vertex $(5,-3)$, opens up (wider than $y=x^2$).}\fi
\end{enumerate}
\vspace{4pt}\hrule\vspace{5pt}
\textbf{Summary (fill in):} In vertex form $y=a(x-h)^2+k$, the vertex is always \bl[2.5cm]{$(h,k)$}, remembering that the sign of $h$ inside the equation is the \bl[2cm]{opposite} of the value you write in the ordered pair. The sign of $a$ tells direction: positive means opens \bl[1.5cm]{up}, negative means opens \bl[2cm]{down}.
