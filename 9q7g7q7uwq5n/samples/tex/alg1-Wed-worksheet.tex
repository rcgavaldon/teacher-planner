\begin{center}{\Large \textbf{Factoring a Difference of Squares}}\end{center}\vspace{0.25cm}
% Factor the difference of squares
\textbf{Example (Factor the difference of squares):}

$x^{2}-9$ — both are perfect squares ($x^2$ and $3^2$) with a minus, so $x^{2}-9=(x+3)(x-3)$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Factor each. If it is not a difference of squares, write “prime.”}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle x^{2}-25$
  \item $\displaystyle x^{2}-49$
  \item $\displaystyle 4x^{2}-1$
  \item $\displaystyle 9x^{2}-16$
  \item $\displaystyle x^{2}+16$
\end{enumerate}
\end{multicols}

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Confirm both terms are perfect squares with a minus between them.
  \item Take the square root of each term ($a$ and $b$).
  \item Write $(a+b)(a-b)$; a sum of squares is prime.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item $(x+5)(x-5)$
  \item $(x+7)(x-7)$
  \item $(2x+1)(2x-1)$
  \item $(3x+4)(3x-4)$
  \item prime (sum of squares)
  \item $(4x+9)(4x-9)$
\end{enumerate}
