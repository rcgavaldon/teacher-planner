\begin{center}{\Large \textbf{Domain and Range}}\end{center}\vspace{0.25cm}
% Domain from an equation
\textbf{Example (Domain from an equation):}

$f(x)=\sqrt{x-2}$ needs $x-2\ge 0$, so the domain is $x\ge 2$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: State the domain. Show the inequality you solved.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle f(x)=\dfrac{1}{x-5}$
  \item $\displaystyle f(x)=\sqrt{x+3}$
  \item $\displaystyle f(x)=x^{2}+1$
  \item $\displaystyle f(x)=\dfrac{x}{x^{2}-9}$
  \item $\displaystyle f(x)=\sqrt{10-x}$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $f(x)=\dfrac{\sqrt{x-1}}{x-4}$ — give the domain (two restrictions).
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Set each denominator $\neq 0$; solve.
  \item Set each radicand $\ge 0$; solve.
  \item Combine the results for the domain.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item $x\neq 5$
  \item $x\ge -3$
  \item all real numbers
  \item $x\neq 3,\ x\neq -3$
  \item $x\le 10$
  \item $x\ge 1$ and $x\neq 4$
\end{enumerate}
