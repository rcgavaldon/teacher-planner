\begin{center}{\Large \textbf{Symmetry: Even, Odd, or Neither}}\end{center}\vspace{0.25cm}
% Even, odd, or neither
\textbf{Example (Even, odd, or neither):}

$f(x)=x^{3}$ gives $f(-x)=(-x)^{3}=-x^{3}=-f(x)$, so it is \textbf{odd} (origin symmetry).

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Find \$f(-x)\$, then classify as even, odd, or neither.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle f(x)=x^{2}+1$
  \item $\displaystyle f(x)=x^{3}-x$
  \item $\displaystyle f(x)=x^{2}+x$
  \item $\displaystyle f(x)=|x|$
  \item $\displaystyle f(x)=x^{5}$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $f(x)=x^4-3x^2$ — test for even/odd and explain.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Substitute $-x$ for $x$.
  \item Simplify carefully (watch the signs on odd powers).
  \item If it equals $f(x)$ it is even; if it equals $-f(x)$ it is odd; otherwise neither.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item Even
  \item Odd
  \item Neither
  \item Even
  \item Odd
  \item Even (all terms have even powers)
\end{enumerate}
