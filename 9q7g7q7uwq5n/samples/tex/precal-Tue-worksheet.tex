\begin{center}{\Large \textbf{Graphical Transformations}}\end{center}\vspace{0.25cm}
% Describe the transformation
\textbf{Example (Describe the transformation):}

$y=-(x-2)^{2}+5$ starts from $y=x^{2}$, moves \textbf{right 2}, \textbf{up 5}, and \textbf{reflects over the x-axis} (because of the $-$).

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: Name the parent and describe every shift, stretch, or reflection.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=2cm]
  \item $\displaystyle y=(x-4)^{2}$
  \item $\displaystyle y=x^{2}+3$
  \item $\displaystyle y=-x^{2}$
  \item $\displaystyle y=|x+2|-1$
  \item $\displaystyle y=\sqrt{x-5}$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=6, itemsep=2.6cm]
  \item $y=-2|x-1|+4$ — describe all transformations.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Identify the parent function.
  \item Rewrite as $a\,f(x-h)+k$.
  \item Read $h$ (opposite sign), $k$, and the sign/size of $a$.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item Quadratic; right 4
  \item Quadratic; up 3
  \item Quadratic; reflect over the x-axis
  \item Absolute value; left 2, down 1
  \item Square root; right 5
  \item Absolute value; left 1, reflect over x-axis, vertical stretch by 2, up 4
\end{enumerate}
