\begin{center}{\Large \textbf{Maximums, Minimums, Increasing and Decreasing}}\end{center}\vspace{0.25cm}
% Read max/min and intervals
\textbf{Example (Read max/min and intervals):}

$y=x^{2}-4$ has a \textbf{minimum} at $(0,-4)$. It \textbf{decreases} on $x<0$ and \textbf{increases} on $x>0$.

\vspace{0.3cm}\hrule\vspace{0.4cm}
\textbf{Directions: State whether there is a max or min, give the point, and the increasing/decreasing intervals.}

\begin{multicols}{2}
\begin{enumerate}[label=\textbf{\arabic*.}, start=1, itemsep=3cm]
  \item $\displaystyle y=x^{2}+2$
  \item $\displaystyle y=-x^{2}+5$
  \item $\displaystyle y=(x-3)^{2}$
  \item $\displaystyle y=-(x+1)^{2}-2$
\end{enumerate}
\end{multicols}

\begin{center}{\large \textbf{Challenge Problems}}\end{center}\vspace{0.3cm}
\begin{multicols}{2}\begin{enumerate}[label=\textbf{\arabic*.}, start=5, itemsep=2.6cm]
  \item $y=(x-2)^2-9$ — give the minimum point and both intervals.
\end{enumerate}\end{multicols}\newpage

\begin{center}{\Large \textbf{Answer Key and Procedure}}\end{center}
\vspace{0.3cm}
\textbf{Procedure}
\begin{enumerate}
  \item Find the vertex (turning point).
  \item Decide max (opens down) or min (opens up).
  \item Read increasing/decreasing on each side of the vertex.
\end{enumerate}
\vspace{0.4cm}\hrule\vspace{0.4cm}
\textbf{Answers}
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item Min $(0,2)$; decreasing $x<0$, increasing $x>0$
  \item Max $(0,5)$; increasing $x<0$, decreasing $x>0$
  \item Min $(3,0)$; decreasing $x<3$, increasing $x>3$
  \item Max $(-1,-2)$; increasing $x<-1$, decreasing $x>-1$
  \item Min $(2,-9)$; decreasing $x<2$, increasing $x>2$
\end{enumerate}
