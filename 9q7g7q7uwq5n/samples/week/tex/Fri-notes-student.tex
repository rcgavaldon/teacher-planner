\begin{center}{\large\textbf{Symmetry: Even \& Odd Functions}}\hfill\ifkey\textcolor{blue!65!black}{\textbf{TEACHER KEY}}\else Name: \rule{3.5cm}{0.4pt}\fi\end{center}
{\small Pre-Calculus \textbar\ Guided notes --- we fill these in together}\vspace{2pt}\hrule\vspace{7pt}
\textbf{Vocabulary}\\[5pt]
\textbf{Even function:} symmetric across the \bl[1.8cm]{$y$-axis}; algebraically $f(-x)=$ \bl[1.6cm]{$f(x)$}.\\[10pt]
\textbf{Odd function:} symmetric about the \bl[1.8cm]{origin}; algebraically $f(-x)=$ \bl[1.8cm]{$-f(x)$}.\\[12pt]
\textbf{We do together.}\quad Is $f(x)=x^{2}-3$ even, odd, or neither?\\[6pt]
\begin{enumerate}[label=\textbf{Step \arabic*.},leftmargin=3.1em,itemsep=9pt]
  \item Replace $x$ with $-x$: $f(-x)=(-x)^{2}-3=$ \{\{$x^{2}-3$|2cm\}\}.
  \item Compare to $f(x)$: they are \bl[1.6cm]{equal}, so $f(-x)=f(x)$.
  \item Therefore $f$ is \bl[1.4cm]{even} (symmetric across the $y$-axis).
\end{enumerate}\vspace{8pt}
\textbf{See it (even = y-axis mirror):}
\begin{center}\begin{tikzpicture}
\begin{axis}[axis lines=middle,xlabel=$x$,ylabel=$y$,xmin=-3.5,xmax=3.5,ymin=-4,ymax=7,width=10cm,height=5.2cm,grid=both,samples=80]
\addplot[very thick,blue,domain=-3:3]{x^2-3};
\end{axis}\end{tikzpicture}\end{center}
{\small\centering $y=x^{2}-3$ --- the $y$-axis is a mirror line (even).\par}
\vspace{6pt}
\textbf{You Try.}\ Solve each. Show every step, then name the answer.\\[6pt]
\begin{enumerate}[label=\textbf{\alph*)},leftmargin=2.4em,itemsep=1.4cm]
  \item Is $f(x)=x^{3}$ even, odd, or neither? \hfill \ifkey\textcolor{blue!65!black}{odd: $f(-x)=-x^{3}=-f(x)$}\fi
  \item Is $f(x)=x^{2}+x$ even, odd, or neither? \hfill \ifkey\textcolor{blue!65!black}{neither}\fi
\end{enumerate}
\vspace{4pt}\hrule\vspace{5pt}
\textbf{Summary (fill in):} Compute $f(-x)$: equals $f(x)\Rightarrow$ \bl[1.2cm]{even}; equals $-f(x)\Rightarrow$ \bl[1.2cm]{odd}; otherwise \bl[1.6cm]{neither}.
